// Inventory: add a product and adjust its stock
function addInventoryItem(name, quantity) {
  const item = {
    id: makeId(),
    name: name.trim(),
    quantity: Number(quantity)
  };
  inventory.push(item);
  saveData('lj-inventory-v1', inventory);
  renderInventory();
}

function changeStock(id, amount) {
  const item = inventory.find(item => item.id === id);
  if (!item) return;
  item.quantity = Math.max(0, item.quantity + amount);
  saveData('lj-inventory-v1', inventory);
  renderInventory();
}

// Task manager: save a task, toggle its status
function addTask(title, priority) {
  tasks.unshift({ id: makeId(), title, priority, done: false });
  saveData('lj-tasks-v1', tasks);
  renderTasks();
}

function toggleTask(id) {
  const task = tasks.find(task => task.id === id);
  if (!task) return;
  task.done = !task.done;
  saveData('lj-tasks-v1', tasks);
  renderTasks();
}

grades.js · core calculation
// Grade calculator: calculate a weighted average
function calculateAverage(subjects) {
  const totalUnits = subjects.reduce(
    (sum, subject) => sum + subject.units, 0
  );
  if (totalUnits === 0) return null;
  const weightedSum = subjects.reduce(
    (sum, subject) => sum + subject.grade * subject.units, 0
  );
  return weightedSum / totalUnits;
}

// Demo passing benchmark: 75 / 100
const average = calculateAverage(subjects);
const passed = average !== null && average >= 75;
